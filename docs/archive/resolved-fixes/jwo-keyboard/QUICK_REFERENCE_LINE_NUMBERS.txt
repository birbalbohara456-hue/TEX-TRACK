	// Store the Dashboard focus return ID before opening a menu.
	// This remembers which Dashboard item had focus so we can restore
	// it after pressing Esc to close the menu.
	window.texTrackRememberDashboardFocus = function () {
		const active = document.activeElement;
		if (active instanceof HTMLElement) {
			const id = active.id;
			if (id) {
				window.texTrackDashboardReturnFocusId = id;
				return;
			}
		}
		// Fallback to dashboard root
		window.texTrackDashboardReturnFocusId = 'dashboard-page-root';
	};

	// Restore focus to the Dashboard item that had focus before the menu opened.
	// Used when closing a menu via Escape while the Dashboard is the active page.
	window.texTrackRestoreDashboardFocus = function () {
		const returnId = window.texTrackDashboardReturnFocusId || 'dashboard-page-root';
		if (!returnId) return false;

		let attempts = 0;
		const focusTarget = function () {
			attempts += 1;
			const element = document.getElementById(returnId);
			if (element instanceof HTMLElement && element.offsetParent !== null) {
				element.focus({ preventScroll: true });
				if (document.activeElement === element) return true;
			}
			// If element doesn't exist or focus failed, try dashboard root as fallback
			if (returnId !== 'dashboard-page-root') {
				const fallback = document.getElementById('dashboard-page-root');
				if (fallback instanceof HTMLElement && fallback.offsetParent !== null) {
					fallback.focus({ preventScroll: true });
					if (document.activeElement === fallback) return true;
				}
			}
			if (attempts < 40) window.setTimeout(focusTarget, 35);
			return false;
		};
		window.requestAnimationFrame(function () {
			window.setTimeout(focusTarget, 0);
		});
	};
